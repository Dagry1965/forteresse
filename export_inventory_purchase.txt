import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../../core/prisma/prisma.service';
import { SequencingService } from '../shared/sequencing.service'; // Assurez-vous du chemin
import { PURCHASE_ORDER_STATUS, DOCUMENT_CONFIG } from '../../../../../shared/constants/status.constants';

@Injectable()
export class PurchaseOrderService {
  constructor(
    private prisma: PrismaService,
    private sequencingService: SequencingService
  ) {}

  /**
   * Création d'une commande fournisseur avec référence automatique
   */
  async createOrder(workspaceId: string, data: any) {
    const { supplier_id, expected_date, items } = data;

    // 1. Génération de la référence via le service généralisé (ex: CMD-2026-0001)
    const reference = await this.sequencingService.generateReference(
      workspaceId,
      'PURCHASE_ORDER'
    );

    // 2. Création de la commande et de ses lignes (Items)
    return this.prisma.purchaseOrder.create({
      data: {
        workspace_id: workspaceId,
        supplier_id: supplier_id,
        reference: reference,
        status: PURCHASE_ORDER_STATUS.DRAFT, // On commence en DRAFT selon vos constantes
        expected_date: expected_date ? new Date(expected_date) : null,
        items: {
          create: items.map((item: any) => ({
            stock_item_id: item.stock_item_id,
            quantity: item.quantity,
            unit_cost: item.unit_cost, // Snapshot du prix d'achat convenu
          })),
        },
      },
      include: {
        items: {
          include: {
            stockItem: true, // Pour afficher le nom des pièces côté Front
          },
        },
        supplier: true,
      },
    });
  }

  /**
   * Récupérer toutes les commandes du garage
   */
  async findAll(workspaceId: string) {
    return this.prisma.purchaseOrder.findMany({
      where: { workspace_id: workspaceId },
      include: {
        supplier: true,
        _count: {
          select: { items: true },
        },
      },
      orderBy: { created_at: 'desc' },
    });
  }

  /**
   * Récupérer une commande précise avec ses détails
   */
  async findOne(workspaceId: string, id: string) {
    const order = await this.prisma.purchaseOrder.findFirst({
      where: {
        id: id,
        workspace_id: workspaceId,
      },
      include: {
        supplier: true,
        items: {
          include: {
            stockItem: true,
          },
        },
        receipts: true, // Pour voir l'historique des réceptions
      },
    });

    if (!order) throw new NotFoundException('Commande fournisseur introuvable');
    return order;
  }

  /**
   * Mettre à jour le statut (ex: passer de DRAFT à SENT)
   */
  async updateStatus(workspaceId: string, id: string, status: string) {
    return this.prisma.purchaseOrder.updateMany({
      where: {
        id: id,
        workspace_id: workspaceId,
      },
      data: { status },
    });
  }
}

import { Injectable } from '@nestjs/common';
import { PrismaService } from '../../core/prisma/prisma.service';

@Injectable()
export class StockService {
  constructor(private readonly prisma: PrismaService) {}

  // ---------------------------------------------------------
  // GET STOCK FOR ONE PRODUCT
  // ---------------------------------------------------------
  async getStock(productId: string) {
    const movements = await this.prisma.stockMovement.findMany({
      where: { product_id: productId },
    });

    const total = movements.reduce(
      (sum, m) => sum + (m.type === 'IN' ? m.quantity : -m.quantity),
      0,
    );

    return { product_id: productId, stock: total };
  }

  // ---------------------------------------------------------
  // LIST MOVEMENTS
  // ---------------------------------------------------------
  async listMovements(workspaceId?: string) {
    return this.prisma.stockMovement.findMany({
      where: workspaceId ? { workspace_id: workspaceId } : undefined,
      orderBy: { created_at: 'desc' },
    });
  }

  // ---------------------------------------------------------
  // REGISTER MOVEMENT
  // ---------------------------------------------------------
  async registerMovement(dto: {
    product_id: string;
    workspace_id: string;
    quantity: number;
    type: 'IN' | 'OUT';
    reason?: string;
  }) {
    return this.prisma.stockMovement.create({
      data: {
        product_id: dto.product_id,
        workspace_id: dto.workspace_id,
        quantity: dto.quantity,
        type: dto.type,
      },
    });
  }
}

import { Injectable, BadRequestException, NotFoundException } from '@nestjs/common';

@Injectable()
export class PurchaseOrderService {
  constructor(private readonly prisma: PrismaService) {}

  async create(workspaceId: string, data: any) {
    return this.prisma.purchaseOrder.create({
      data: {
        workspace_id: workspaceId,
        supplier_id: data.supplierId,
        reference: data.reference || `PO-${Date.now()}`,
        status: PURCHASE_ORDER_STATUS.DRAFT,
        items: {
          create: (data.items || []).map((line: any) => ({
            item_id: line.itemId,
            quantity: Number(line.quantity),
            price_buy: Number(line.price_buy),

      },
    });

    return this.prisma.purchaseOrder.findMany({
      where: { workspace_id: workspaceId },
      include: { items: { include: { item: true } }, supplier: true },
      orderBy: { created_at: 'desc' }
    });
  }

  async findOne(workspaceId: string, id: string) {
    return this.prisma.purchaseOrder.findFirst({
      where: { id, workspace_id: workspaceId },
      include: { items: { include: { item: true } }, supplier: true },
    });
  }
}

import { Injectable, BadRequestException, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../../core/prisma/prisma.service';
import { PURCHASE_ORDER_STATUS, STOCK_MOVEMENT_TYPE } from '../../../../../shared/constants/status.constants';

@Injectable()
export class PurchaseReceiptService {
  constructor(private readonly prisma: PrismaService) {}

  async createReceipt(workspaceId: string, data: {
    reference: string; // Numéro de Bon de Livraison
    purchase_order_id: string;
    items: { item_id: string; quantity: number }[];
  }) {
    return this.prisma.$transaction(async (tx) => {
      // 1. Créer le Bon de Réception
      const receipt = await tx.purchaseReceipt.create({
        data: {
          reference: data.reference,
          workspace_id: workspaceId,
          purchase_order_id: data.purchase_order_id,
        },
      });

      // 2. Traiter chaque article reçu
      for (const receivedItem of data.items) {
        // A. Trouver la ligne de commande correspondante
        const orderLine = await tx.purchaseOrderItem.findFirst({
          where: {
            purchase_order_id: data.purchase_order_id,
            item_id: receivedItem.item_id,
          },
        });

        if (!orderLine) throw new BadRequestException(`Article ${receivedItem.item_id} non présent dans la commande.`);

        // B. Mettre à jour la quantité reçue sur la commande
        await tx.purchaseOrderItem.update({
          where: { id: orderLine.id },
          data: { received_quantity: { increment: receivedItem.quantity } },
        });

        // C. Augmenter le stock réel dans StockItem
        await tx.stockItem.update({
          where: { id: receivedItem.item_id },
          data: { quantity: { increment: receivedItem.quantity } },
        });

        // D. Créer le mouvement de stock (Historique)
        await tx.stockMovement.create({
          data: {
            workspace_id: workspaceId,
            item_id: receivedItem.item_id,
            quantity: receivedItem.quantity,
            type: STOCK_MOVEMENT_TYPE.IN_PURCHASE,
          },
        });
      }

      // 3. Mettre à jour le statut de la commande (Partiel ou Complet)
      const fullOrder = await tx.purchaseOrder.findUnique({
        where: { id: data.purchase_order_id },
        include: { items: true }
      });

      const isFullyReceived = fullOrder.items.every(i => i.received_quantity >= i.quantity);
      
      await tx.purchaseOrder.update({
        where: { id: data.purchase_order_id },
        data: { 
          status: isFullyReceived ? PURCHASE_ORDER_STATUS.RECEIVED : PURCHASE_ORDER_STATUS.PARTIALLY_RECEIVED 
        }
      });

      return receipt;
    });
  }
}
